-- Database: team_palinoia
CREATE DATABASE IF NOT EXISTS team_palinoia;
USE team_palinoia;


-- 1. team_members (login credentials + name, includes admin)
CREATE TABLE team_members (
    member_id INT AUTO_INCREMENT PRIMARY KEY,
    name      VARCHAR(100) NOT NULL,
    email     VARCHAR(150) NOT NULL UNIQUE,
    password  VARCHAR(100) NOT NULL,
    role      ENUM('member', 'admin') DEFAULT 'member'
);

-- 2. tasks (assigned to a team member)
CREATE TABLE tasks (
    task_id     INT AUTO_INCREMENT PRIMARY KEY,
    member_id   INT NOT NULL,
    title       VARCHAR(200) NOT NULL,
    description TEXT,
    status      ENUM('urgent', 'in_progress', 'completed') DEFAULT 'in_progress',
    deadline    DATE,
    event_name  VARCHAR(150),
    FOREIGN KEY (member_id) REFERENCES team_members(member_id)
);
